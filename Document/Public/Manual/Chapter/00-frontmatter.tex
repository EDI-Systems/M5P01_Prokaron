\chapter*{System Features} \addcontentsline{toc}{chapter}{System Features} \markboth{System Features}{}

\begin{center} {\sffamily\large\color{rmpgray}M5P01 (Prokaron) R6T1}\\[2pt] {\sffamily\large\bfseries\color{rmpgray}Light-Weight Real-Time Operating System\\[2pt](Sixth Edition, Type One)}\\[4pt] {\rmfamily\color{rmpmid}Technical Reference Manual}\\[12pt] \end{center}

\begin{description}[leftmargin=0pt,labelindent=0pt,style=nextline,itemsep=8pt]
    \item[\textsf{\color{rmppurple}High usability}]
    Light-weight and easy to use as first-class requirements.
    Easy to configure the development environment and the project paths.
    Provides the most common inter-thread communication interfaces.

    \item[\textsf{\color{rmppurple}Hard real-time}]
    A fully preemptive hard-real-time kernel.
    Round-robin scheduling between threads of the same priority.
    All key operations have O(1) time complexity.

    \item[\textsf{\color{rmppurple}High portability}]
    The least assembly language among all RTOSes.
    Written to ANSI C89 and MISRA C requirements for compiler compatibility.
    Extremely low resource requirements for a minimal system.
    Can run as a guest operating system on another operating system.

    \item[\textsf{\color{rmppurple}High efficiency}]
    Performance-critical code is carefully optimised.
    Extremely short thread switch time and interrupt response time.

    \item[\textsf{\color{rmppurple}High reliability}]
    Provides parameter assertion and pervasive interface-parameter checking.
    Formally verified to the equivalent of IEC 61508 SIL 4.\footnote{This work is not complete. The current software is equivalent to a SIL 2 pre-certification level.}
\end{description}
